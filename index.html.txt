function normalize(str) {
  return (str || "").toString().normalize("NFD").replace(/[\u0300-\u036f]/g, "").toUpperCase().trim();
}

function extractPhones(text) {
  if (!text) return [];
  const matches = text.match(/(?:2\d{9}|1535|1566|14500|14960|14741)/g);
  return matches ? Array.from(new Set(matches)) : [];
}

function extractSchedule(text) {
  if (!text) return null;
  const match = text.match(/(?:Έκδοση Προγράμματος|Ενημέρωση Προγράμματος|Πρόγραμμα|αναρτώνται|δίνονται|ανοίγουν)\s*:?\s*([^.–\n]+)/i);
  return match ? match[1].trim() : null;
}

let database = RAW_SNAPSHOT.map(item => ({
  ...item,
  phones: extractPhones(item.detail),
  scheduleRule: extractSchedule(item.detail)
}));

const STORAGE_KEY = "callcenter_custom_db_v3";
let customEntries = JSON.parse(localStorage.getItem(STORAGE_KEY) || "[]");

function getAllRecords() {
  return [...customEntries, ...database];
}

function smartSearch(queryText) {
  const q = normalize(queryText);
  if (!q) return [];

  const rawWords = q.split(/\s+/).filter(w => w.length > 1);
  const filterWords = rawWords.filter(w => !["ΠΟΤΕ","ΑΝΟΙΓΟΥΝ","ΤΑ","ΡΑΝΤΕΒΟΥ","ΣΤΟ","ΣΤΟΝ","ΣΤΗΝ","ΓΙΝΕΤΑΙ","ΕΧΕΙ","ΓΙΑ"].includes(w));

  return getAllRecords().filter(r => {
    const hospNorm = normalize(r.hospital);
    const examNorm = normalize(r.exam);
    const detailNorm = normalize(r.detail);
    const ypeNorm = normalize(r.ype);

    const fullRecordText = `${hospNorm} ${examNorm} ${detailNorm} ${ypeNorm}`;
    return filterWords.every(word => fullRecordText.includes(word));
  });
}

function render() {
  const contentEl = document.getElementById("content");
  const query = document.getElementById("search").value;
  const results = smartSearch(query);

  if (!query.trim()) {
    contentEl.innerHTML = `
      <div class="empty">
        Πληκτρολογήστε ερώτημα όπως μιλάει ο πολίτης.
        <div class="hints">
          <button class="hint-btn" onclick="setQuery('πότε ανοίγουν τα ραντεβού Ιπποκράτειο Θεσσαλονίκης')">Ιπποκράτειο Θεσσαλονίκης (Ραντεβού)</button>
          <button class="hint-btn" onclick="setQuery('πότε ανοίγουν τα ραντεβού Ιπποκράτειο Αθήνας')">Ιπποκράτειο Αθήνας (Ραντεβού)</button>
          <button class="hint-btn" onclick="setQuery('σπινθηρογράφημα Βόλος')">Σπινθηρογράφημα Βόλος</button>
        </div>
      </div>`;
    return;
  }

  if (results.length === 0) {
    contentEl.innerHTML = `<div class="empty">Δεν βρέθηκαν αποτελέσματα. Δοκιμάστε με πιο απλούς όρους.</div>`;
    return;
  }

  contentEl.innerHTML = `
    <div style="font-size:12px; color:var(--muted); margin-bottom:10px;">Βρέθηκαν ${results.length} ακριβείς καταχωρήσεις</div>
    ${results.map(r => renderCard(r)).join('')}
  `;
}

function renderCard(r) {
  const phonesHtml = r.phones.length 
    ? r.phones.map(p => `<a href="tel:${p}" class="phone-link">📞 ${p}</a>`).join('')
    : '<span style="color:var(--muted); font-size:12px;">Δεν αναγράφεται απευθείας τηλέφωνο</span>';

  return `
    <div class="card">
      <div class="card-header">
        <div>
          <div class="hosp-name">${r.hospital}</div>
          <div style="font-size:12px; color:var(--muted); margin-top:2px;">${r.ype || ''}</div>
        </div>
        ${r.exam ? `<span class="type-tag">${r.exam}</span>` : `<span class="type-tag" style="background:#EBF5F0;color:var(--primary)">Γενικές Πληροφορίες</span>`}
      </div>

      <div class="info-grid">
        <div class="info-box">
          <strong>Τηλέφωνα Ραντεβού / Τμήματος</strong>
          <div>${phonesHtml}</div>
        </div>
        ${r.scheduleRule ? `
        <div class="info-box" style="background:#FFFCE8; border-color:#F5E6A3;">
          <strong style="color:#8A6D0B;">Πότε Ανοίγουν τα Ραντεβού</strong>
          <span style="font-weight:600; font-size:13px; color:#1E3932;">${r.scheduleRule}</span>
        </div>` : ''}
      </div>

      <div class="detail-text">
        ${r.detail}
      </div>
    </div>
  `;
}

function setQuery(text) {
  document.getElementById("search").value = text;
  render();
}

document.getElementById("search").addEventListener("input", render);

  if (!formOpen) { formHolder.innerHTML = ""; return; }

    </div>`;

function saveEntry() {
  const hosp = document.getElementById("f_hosp").value.trim();
  const exam = document.getElementById("f_exam").value.trim();
  const ype = document.getElementById("f_ype").value.trim();
  const detail = document.getElementById("f_detail").value.trim();

  if (!hosp) return;

  const newObj = {
    id: Date.now().toString(),
    hospital: hosp,
    exam: exam,
    ype: ype,
    detail: detail,
    phones: extractPhones(detail),
    scheduleRule: extractSchedule(detail),
    custom: true
  };

  customEntries.unshift(newObj);
  localStorage.setItem(STORAGE_KEY, JSON.stringify(customEntries));
  formOpen = false;
  formHolder.innerHTML = "";
  render();
}
